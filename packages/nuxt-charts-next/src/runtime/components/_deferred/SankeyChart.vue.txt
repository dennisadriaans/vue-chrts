<script setup lang="ts" generic="N extends SankeyInputNode = SankeyInputNode, L extends SankeyInputLink = SankeyInputLink">
/**
 * Sankey chart adapter.
 *
 * Accepts the `nuxt-charts` v2 `SankeyChartProps` config API and renders the
 * `vccs` `<Sankey>`. `vccs` requires explicit numeric `width`/`height`, so the
 * chart is wrapped in `<ResponsiveContainer>` which clones those dimensions in.
 */
import { computed } from "vue";
import { ResponsiveContainer, Sankey, Tooltip } from "vccs";
import type {
  SankeyChartProps,
  SankeyInputLink,
  SankeyInputNode,
} from "../types/charts";

const props = defineProps<SankeyChartProps<N, L>>();

const margin = computed(() => ({
  top: props.padding?.top ?? 5,
  right: props.padding?.right ?? 5,
  bottom: props.padding?.bottom ?? 5,
  left: props.padding?.left ?? 5,
}));
</script>

<template>
  <ResponsiveContainer width="100%" :height="height">
    <Sankey
      :data="data"
      :node-padding="nodePadding ?? 10"
      :node-width="nodeWidth ?? 10"
      :iterations="iterations ?? 32"
      :margin="margin"
    >
      <template v-if="$slots.node" #node="slotProps">
        <slot name="node" v-bind="slotProps" />
      </template>
      <template v-if="$slots.link" #link="slotProps">
        <slot name="link" v-bind="slotProps" />
      </template>
      <Tooltip v-if="!hideTooltip" />
    </Sankey>
  </ResponsiveContainer>
</template>
